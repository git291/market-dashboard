async function loadData() { try { const res = await fetch('data.json?t=' + Date.now()); const data = await res.json(); if (data.updated_at) { document.getElementById('updateTime').innerText = `更新時間：${data.updated_at}`; const rawDateStr = data.updated_at.split(' ')[0]; const datePart = rawDateStr.replace(/-/g, '/'); const d = new Date(rawDateStr); const dayMap = ['日', '一', '二', '三', '四', '五', '六']; const dayName = isNaN(d.getDay()) ? '' : ` (${dayMap[d.getDay()]})`; document.getElementById('todayBadge').innerText = `📅 ${datePart}${dayName}`; } const renderTopCard = (id, item) => { if (!item || item.price === '--') return; document.getElementById(`${id}-price`).innerText = item.price; document.getElementById(`${id}-prev`).innerText = item.prev_close; const weekEl = document.getElementById(`${id}-week`); if (weekEl) { weekEl.innerText = item.week_pChange; const isUp = item.week_pChange.includes('+'); weekEl.className = isUp ? 'red' : 'green'; } }; renderTopCard('twii', data.twii); renderTopCard('tsmc', data.tsmc); renderTopCard('etf6208', data.etf6208); const renderCard = (id, item, forceRed = false) => { if (!item || item.price === '--') return; const priceEl = document.getElementById(`${id}-price`); const changeEl = document.getElementById(`${id}-change`); if (priceEl) { priceEl.innerText = item.price; const isUp = item.raw_change > 0; const colorClass = forceRed ? 'red' : (isUp ? 'red' : 'green'); priceEl.className = `${priceEl.className.replace(/red|green/g, '')} ${colorClass}`; } if (changeEl) { const isUp = item.raw_change > 0; const colorClass = forceRed ? 'red' : (isUp ? 'red' : 'green'); const arrow = (isUp || forceRed) ? '▲' : '▼'; changeEl.innerText = `${arrow} ${Math.abs(item.change)} (${item.pChange})`; changeEl.className = `${changeEl.className.replace(/red|green/g, '')} ${colorClass}`; } }; ['dji', 'ixic', 'sox', 'vix', 'brent', 'fitx'].forEach(k => renderCard(k, data[k])); renderCard('usdtwd', data.usdtwd, true); // 1. 動態更新 VIX 的開高低收細節 if (data.vix) { document.getElementById('vix-open').innerText = data.vix.open; document.getElementById('vix-high').innerText = data.vix.high; document.getElementById('vix-low').innerText = data.vix.low; document.getElementById('vix-prev').innerText = data.vix.prev; } // 2. 動態更新 美國10年期公債殖利率 if (data.bond) { document.getElementById('bond-price').innerText = `${data.bond.price} %`; document.getElementById('bond-prev').innerText = `${data.bond.prev_close} %`; } // 3. 動態渲染 資券進出行情表格 if (data.margin_data && data.margin_data.length > 0) { const tbody = document.querySelector('.etf-table tbody'); tbody.innerHTML = ''; data.margin_data.forEach(row => { const marginClass = row.margin_buy_sell.includes('-') ? 'green' : 'red'; const shortClass = row.short_buy_sell.includes('-') ? 'green' : 'red'; const tr = document.createElement('tr'); tr.innerHTML = ` ${row.date} ${row.margin_buy_sell} ${row.short_buy_sell} ${row.margin_balance} ${row.short_balance} `; tbody.appendChild(tr); }); } if (data.fear) { const score = parseInt(data.fear.score) || 31; document.getElementById('fear-score').innerText = score; const angle = (score / 100) * Math.PI; document.getElementById('needle').setAttribute('x2', (50 - 32 * Math.cos(angle)).toFixed(1)); document.getElementById('needle').setAttribute('y2', (50 - 32 * Math.sin(angle)).toFixed(1)); } if (data.charts) { renderDetailChart('chart-twii', data.charts.twii, data.twii?.raw_change >= 0 ? '#ef4444' : '#00e676'); renderDetailChart('chart-fitx', data.charts.twii, data.fitx?.raw_change >= 0 ? '#ef4444' : '#00e676'); renderDetailChart('chart-tsmc', data.charts.tsmc, data.tsmc?.raw_change >= 0 ? '#ef4444' : '#00e676'); renderDetailChart('chart-etf6208', data.charts.etf6208, data.etf6208?.raw_change >= 0 ? '#ef4444' : '#00e676'); if (data.charts.twd) { renderDetailChart('twdCanvas', data.charts.twd, '#eab308'); } if (data.charts.brent) { renderDetailChart('brentCanvas', data.charts.brent, data.brent?.raw_change >= 0 ? '#ef4444' : '#00e676'); } } } catch (e) { console.error("同步失敗:", e); } }
